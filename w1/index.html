document.addEventListener('DOMContentLoaded', () => {
  // 1. 食物選單清單與對應照片
  const foodList = [
    {
      name: '拉麵 🍜',
      image: 'https://images.unsplash.com/photo-1569718212165-3a8278d5f624?auto=format&fit=crop&w=800&q=80'
    },
    {
      name: '日式炸豬排 🍱',
      image: 'https://images.unsplash.com/photo-1591814468924-caf88d1232e1?auto=format&fit=crop&w=800&q=80'
    },
    {
      name: '燒肉飯 🥩',
      image: 'https://images.unsplash.com/photo-1544025162-d76694265947?auto=format&fit=crop&w=800&q=80'
    },
    {
      name: '土豆絲炒肉 🍲',
      image: 'https://images.unsplash.com/photo-1541832676-9b763b0239ab?auto=format&fit=crop&w=800&q=80'
    },
    {
      name: '豚骨拉麵 🍜',
      image: 'https://images.unsplash.com/photo-1591814468924-caf88d1232e1?auto=format&fit=crop&w=800&q=80'
    }
  ];

  // 2. 取得 DOM 元素
  const mainTitle = document.getElementById('mainTitle');
  const drawBtn = document.getElementById('drawBtn');
  const foodImage = document.getElementById('foodImage');
  const placeholderIcon = document.getElementById('placeholderIcon');
  const mapBtn = document.getElementById('mapBtn');

  // 3. 綁定點擊事件
  drawBtn.addEventListener('click', () => {
    // 快速滾動抽籤效果
    let count = 0;
    const interval = setInterval(() => {
      const randomIndex = Math.floor(Math.random() * foodList.length);
      mainTitle.textContent = foodList[randomIndex].name;
      count++;

      if (count > 10) {
        clearInterval(interval);
        
        // 最終隨機結果
        const finalChoice = foodList[Math.floor(Math.random() * foodList.length)];
        
        // 替換標題與觸發彈跳動畫
        mainTitle.textContent = finalChoice.name;
        mainTitle.classList.remove('bounce-anim');
        void mainTitle.offsetWidth; // 觸發重繪 (Reflow)
        mainTitle.classList.add('bounce-anim');

        // 更新圖片
        foodImage.src = finalChoice.image;
        foodImage.style.display = 'block';
        placeholderIcon.style.display = 'none';

        // 更新 Google Maps 導航連結
        const rawName = finalChoice.name.replace(/[^\u4e00-\u9fa5a-zA-Z]/g, ''); // 移除表情符號
        mapBtn.href = `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(rawName)}`;
        mapBtn.classList.add('show');
      }
    }, 80);
  });
});
